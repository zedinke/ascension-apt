sudo curl -fsSLo /usr/share/keyrings/ascension-archive-keyring.gpg \
     https://zedinke.github.io/ascension-apt/ascension-archive-keyring.gpg

sudo tee /etc/apt/sources.list.d/ascension.sources <<'EOF2'
Types: deb
URIs: https://zedinke.github.io/ascension-apt
Suites: apex
Components: main apps
Architectures: amd64
Signed-By: /usr/share/keyrings/ascension-archive-keyring.gpg
EOF2

sudo apt update
